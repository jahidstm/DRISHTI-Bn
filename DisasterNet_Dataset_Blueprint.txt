import pandas as pd
import time
from deep_translator import GoogleTranslator
from tqdm import tqdm

def batch_translation_with_checkpointing(df_source, chunk_size=100):
    translator = GoogleTranslator(source='en', target='bn')
    target_csv = 'DisasterNet_Bangla_Dataset_Final.csv'
    pd.DataFrame(columns=['image_path', 'en_caption', 'bn_caption', 'severity_label']).to_csv(target_csv, index=False)
    
    for start in tqdm(range(0, len(df_source), chunk_size), desc="Batch Processing"):
        chunk = df_source.iloc[start:start + chunk_size]
        translated_chunk = []
        
        for _, row in chunk.iterrows():
            try:
                bn_text = translator.translate(row['en_caption'])
                translated_chunk.append({'image_path': row['image_path'], 'en_caption': row['en_caption'], 'bn_caption': bn_text, 'severity_label': row['severity_label']})
                time.sleep(1) # Evasion tactic
            except Exception as e:
                translated_chunk.append({'image_path': row['image_path'], 'en_caption': row['en_caption'], 'bn_caption': '[FAILED]', 'severity_label': row['severity_label']})
                time.sleep(5)
        
        pd.DataFrame(translated_chunk).to_csv(target_csv, mode='a', header=False, index=False)
